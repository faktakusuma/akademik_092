## 1. PENDAHULUAN & LATAR BELAKANG

Dalam pengembangan perangkat lunak berbasis data relasional, fase perancangan skema fisik (*Physical Database Schema*) memegang peranan krusial sebagai fondasi integritas, performa, dan keamanan informasi. Berdasarkan hasil pemodelan konseptual dan normalisasi tingkat ketiga (3NF) yang telah dilakukan pada modul sebelumnya, perancangan tersebut diterjemahkan ke dalam bahasa formal mesin basis data melalui *Data Definition Language* (DDL). 

Modul 5 berfokus pada implementasi nyata perintah DDL (`CREATE TABLE`, `ALTER TABLE`, `DROP TABLE`) ke dalam sistem manajemen basis data MySQL melalui antarmuka phpMyAdmin. Tidak hanya sekadar membuat tabel, perancangan ini menerapkan aturan integritas data secara ketat menggunakan *Constraints* (`PRIMARY KEY`, `FOREIGN KEY`, `UNIQUE`, `NOT NULL`, `CHECK`, dan `DEFAULT`) guna mencegah masuknya data anomali atau tidak valid.

## 3. IMPLEMENTASI KODE SQL DDL

Berikut adalah naskah skrip SQL komprehensif yang dieksekusi di dalam basis data `db_akademik_baru_25430092`:

```sql
-- Pembuatan dan pemilihan basis data utama proyek
CREATE DATABASE IF NOT EXISTS db_akademik_baru_25430092 
    CHARACTER SET utf8mb4 
    COLLATE utf8mb4_general_ci;
USE db_akademik_baru_25430092;

-- Menonaktifkan pemeriksaan kunci asing sementara untuk pembersihan tabel berelasi
SET foreign_key_checks = 0;

DROP TABLE IF EXISTS krs;
DROP TABLE IF EXISTS kelas_kuliah;
DROP TABLE IF EXISTS mata_kuliah;
DROP TABLE IF EXISTS mahasiswa;
DROP TABLE IF EXISTS dosen;

-- Mengaktifkan kembali pemeriksaan integritas kunci asing
SET foreign_key_checks = 1;

-- 1. Pembuatan Tabel Dosen
CREATE TABLE dosen (
    id_dosen INT NOT NULL AUTO_INCREMENT,
    nidn_dosen CHAR(10) NOT NULL,
    nama_dosen VARCHAR(100) NOT NULL,
    email_dosen VARCHAR(100) NOT NULL,
    PRIMARY KEY (id_dosen),
    UNIQUE KEY uq_dosen_nidn (nidn_dosen)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. Pembuatan Tabel Mahasiswa
CREATE TABLE mahasiswa (
    id_mhs INT NOT NULL AUTO_INCREMENT,
    nim_mhs CHAR(10) NOT NULL,
    nama_mhs VARCHAR(100) NOT NULL,
    prodi_mhs VARCHAR(50) NOT NULL,
    angkatan_mhs YEAR NOT NULL,
    status_mhs ENUM('aktif', 'cuti', 'lulus') NOT NULL DEFAULT 'aktif',
    PRIMARY KEY (id_mhs),
    UNIQUE KEY uq_mhs_nim (nim_mhs)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 3. Pembuatan Tabel Mata Kuliah dengan Validasi Check Constraint SKS
CREATE TABLE mata_kuliah (
    id_mk INT NOT NULL AUTO_INCREMENT,
    kode_mk VARCHAR(10) NOT NULL,
    nama_mk VARCHAR(100) NOT NULL,
    sks_mk TINYINT NOT NULL,
    semester_mk TINYINT NOT NULL,
    PRIMARY KEY (id_mk),
    UNIQUE KEY uq_mk_kode (kode_mk),
    CONSTRAINT ck_mk_sks CHECK (sks_mk > 0 AND sks_mk <= 6)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 4. Pembuatan Tabel Kelas Kuliah dengan Relasi ke Dosen dan Mata Kuliah
CREATE TABLE kelas_kuliah (
    id_kelas INT NOT NULL AUTO_INCREMENT,
    nama_kelas VARCHAR(20) NOT NULL,
    id_mk INT NOT NULL,
    id_dosen INT NOT NULL,
    ruangan VARCHAR(20) NOT NULL,
    PRIMARY KEY (id_kelas),
    CONSTRAINT fk_kelas_mk FOREIGN KEY (id_mk) 
        REFERENCES mata_kuliah (id_mk) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_kelas_dosen FOREIGN KEY (id_dosen) 
        REFERENCES dosen (id_dosen) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 5. Pembuatan Tabel Transaksi KRS dengan Composite Key dan Cascade Delete
CREATE TABLE krs (
    id_mhs INT NOT NULL,
    id_kelas INT NOT NULL,
    nilai_huruf CHAR(2) NULL,
    PRIMARY KEY (id_mhs, id_kelas),
    CONSTRAINT fk_krs_mhs FOREIGN KEY (id_mhs) 
        REFERENCES mahasiswa (id_mhs) ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_krs_kelas FOREIGN KEY (id_kelas) 
        REFERENCES kelas_kuliah (id_kelas) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
